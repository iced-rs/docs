15/// The logic of a [`Transition`].
16pub trait Program: 'static {
17    /// The type of value that the [`Program`] animates.
18    type Value: Copy + 'static;
19
20    /// Transitions the [`Program`] from its current state towards the given value at
21    /// the given time.
22    fn go(&mut self, value: Self::Value, now: Instant);
23
24    /// Returns true if the [`Program`] is currently in progress.
25    fn is_animating(&self, now: Instant) -> bool;
26}
27
28impl<I> Program for Animation<I>
29where
30    I: Float + Clone + Copy + PartialEq + 'static,
31{
32    type Value = I;
33
34    fn go(&mut self, value: Self::Value, now: Instant) {
35        self.go_mut(value, now);
36    }
37
38    fn is_animating(&self, now: Instant) -> bool {
39        self.is_animating(now)
40    }
41}

43/// A widget that can be used to animate its contents.
44pub struct Transition<'a, Message, Theme, Renderer, P>
45where
46    P: Program,
47{
48    init: Box<dyn Fn() -> P + 'a>,
49    view: Box<dyn Fn(&P, Instant) -> Element<'a, Message, Theme, Renderer> + 'a>,
50    on_finish: Option<Box<dyn Fn() -> Message + 'a>>,
51    element: Element<'a, Message, Theme, Renderer>,
52    next_element: Option<Element<'a, Message, Theme, Renderer>>,
53    last_limits: layout::Limits,
54    key: Key,
55    id: Option<widget::Id>,
56    value: P::Value,
57}

59impl<'a, Message, Theme, Renderer, P> Transition<'a, Message, Theme, Renderer, P>
60where
61    Renderer: core::Renderer,
62    P: Program,
63{
64    /// Creates a new [`Transition`].
65    ///
66    /// The `init` closure will be used to initialize an animation.
67    ///
68    /// The `view` closure will receive the animation and an [`Instant`], which can be used for interpolating values.
69    /// This will be called every frame until the given `value` is reached.
70    pub fn new<E>(
71        init: impl Fn() -> P + 'a,
72        value: P::Value,
73        view: impl Fn(&P, Instant) -> E + 'a,
74    ) -> Self
75    where
76        E: Into<Element<'a, Message, Theme, Renderer>>,
77    {
78        Self {
79            init: Box::new(init),
80            view: Box::new(move |program, at| view(program, at).into()),
81            on_finish: None,
82            element: Element::new(space()),
83            next_element: None,
84            last_limits: layout::Limits::new(Size::ZERO, Size::ZERO),
85            key: Key::default(),
86            id: None,
87            value,
88        }
89    }
90
91    /// Sets the [`widget::Id`] of the [`Transition`].
92    ///
93    /// The [`widget::Id`] can subsequently be used to reset the [`Animation`] via [`reset`].
94    pub fn id(mut self, id: impl Into<widget::Id>) -> Self {
95        self.id = Some(id.into());
96        self
97    }
98
99    /// Sets the key of the [`Transition`] widget for continuity.
100    ///
101    /// Changing the key will reset the [`Animation`].
102    pub fn key(mut self, key: impl Hash) -> Self {
103        self.key = Key::new(key);
104        self
105    }
106
107    /// Sets the message that will be produced when the [`Transition`] has finished animating.
108    ///
109    /// Note that if an [`Animation`] is set to loop forever, the message will never be produced!
110    pub fn on_finish(self, on_finish: Message) -> Self
111    where
112        Message: Clone + 'a,
113    {
114        self.on_finish_maybe(on_finish)
115    }
116
117    /// Sets the message that will be produced when the [`Transition`] has finished animating, if [`Some`].
118    ///
119    /// Note that if an [`Animation`] is set to loop forever, the message will never be produced!
120    pub fn on_finish_maybe(mut self, on_finish: impl Into<Option<Message>>) -> Self
121    where
122        Message: Clone + 'a,
123    {
124        self.on_finish = on_finish
125            .into()
126            .map(|on_finish| Box::new(move || on_finish.clone()) as _);
127
128        self
129    }
130
131    /// Sets the message that will be produced when the [`Transition`] has finished animating.
132    ///
133    /// This is analogous to [`Transition::on_finish`], but using a closure to produce
134    /// the message.
135    ///
136    /// This closure will only be called when the [`Transition`] has actually finished animating and,
137    /// therefore, this method is useful to reduce overhead if creating the resulting
138    /// message is slow.
139    ///
140    /// Note that if an [`Animation`] is set to loop forever, the message will never be produced!
141    pub fn on_finish_with(mut self, on_finish: impl Fn() -> Message + 'a) -> Self {
142        self.on_finish = Some(Box::new(on_finish));
143        self
144    }
145}

147impl<Message, Theme, Renderer, P> Widget<Message, Theme, Renderer>
148    for Transition<'_, Message, Theme, Renderer, P>
149where
150    Renderer: core::Renderer,
151    P: Program,
152{
153    fn size(&self) -> Size<Length> {
154        self.element.as_widget().size()
155    }
156
157    fn tag(&self) -> tree::Tag {
158        tree::Tag::of::<State<P>>()
159    }
160
161    fn state(&self) -> tree::State {
162        tree::State::new(State {
163            animation: (self.init)(),
164            instant: Instant::now(),
165            key: self.key,
166            should_reset: false,
167            size: None,
168        })
169    }
170
171    fn diff(&mut self, tree: &mut Tree) {
172        let State::<P> {
173            animation,
174            key: old_key,
175            should_reset,
176            instant,
177            ..
178        } = tree.state.downcast_mut();
179
180        if *old_key != self.key {
181            *old_key = self.key;
182            *animation = (self.init)();
183            *should_reset = false;
184        }
185
186        if let Some(next_element) = self.next_element.take() {
187            self.element = next_element;
188        } else {
189            self.element = (self.view)(animation, *instant);
190        }
191
192        tree.diff_children(std::slice::from_mut(&mut self.element));
193    }
194
195    fn layout(&mut self, tree: &mut Tree, renderer: &Renderer, limits: &layout::Limits) {
196        self.last_limits = *limits;
197
198        self.element
199            .as_widget_mut()
200            .layout(&mut tree.children[0], renderer, &limits.loose());
201
202        tree.size = tree.children[0].size;
203    }
204
205    fn update(
206        &mut self,
207        tree: &mut Tree,
208        event: &Event,
209        layout: Layout,
210        cursor: mouse::Cursor,
211        renderer: &Renderer,
212        shell: &mut Shell<'_, Message>,
213        viewport: &Rectangle,
214    ) {
215        if let core::Event::Window(core::window::Event::RedrawRequested(redraw)) = event {
216            let State::<P> {
217                animation,
218                instant,
219                should_reset,
220                size,
221                ..
222            } = tree.state.downcast_mut();
223
224            if instant == redraw {
225                shell.request_redraw();
226            } else {
227                let was_animating = animation.is_animating(*instant);
228
229                if *should_reset {
230                    *animation = (self.init)();
231                    *should_reset = false;
232                }
233
234                *instant = *redraw;
235                animation.go(self.value, *instant);
236
237                let is_animating = animation.is_animating(*instant);
238                let just_finished = was_animating && !is_animating;
239
240                if is_animating || just_finished {
241                    let size = *size;
242
243                    let mut new = (self.view)(animation, *instant);
244                    tree.diff_children(&mut [new.as_widget_mut()]);
245
246                    let new_size = new.as_widget().size();
247
248                    if size != Some(new_size) {
249                        self.next_element = Some(new);
250                        shell.invalidate_layout_with(shell::Diff::Perform);
251
252                        let state = tree.state.downcast_mut::<State<P>>();
253                        state.size = Some(new_size);
254                    } else {
255                        self.element = new;
256                        self.element.as_widget_mut().layout(
257                            &mut tree.children[0],
258                            renderer,
259                            &self.last_limits,
260                        );
261                    }
262
263                    shell.request_redraw();
264                }
265
266                if just_finished && let Some(on_finish) = &self.on_finish {
267                    shell.publish(on_finish());
268                }
269            }
270        }
271
272        self.element.as_widget_mut().update(
273            &mut tree.children[0],
274            event,
275            layout,
276            cursor,
277            renderer,
278            shell,
279            viewport,
280        );
281    }
282
283    fn draw(
284        &self,
285        tree: &Tree,
286        renderer: &mut Renderer,
287        theme: &Theme,
288        style: &renderer::Style,
289        layout: Layout,
290        cursor: mouse::Cursor,
291        viewport: &Rectangle,
292    ) {
293        self.element.as_widget().draw(
294            &tree.children[0],
295            renderer,
296            theme,
297            style,
298            layout,
299            cursor,
300            viewport,
301        );
302    }
303
304    fn mouse_interaction(
305        &self,
306        tree: &Tree,
307        layout: Layout,
308        cursor: mouse::Cursor,
309        viewport: &Rectangle,
310        renderer: &Renderer,
311    ) -> mouse::Interaction {
312        self.element.as_widget().mouse_interaction(
313            &tree.children[0],
314            layout,
315            cursor,
316            viewport,
317            renderer,
318        )
319    }
320
321    fn operate(
322        &mut self,
323        tree: &mut Tree,
324        layout: Layout,
325        viewport: &Rectangle,
326        renderer: &Renderer,
327        operation: &mut dyn widget::Operation,
328    ) {
329        let mut should_reset = ShouldReset(false);
330        operation.custom(self.id.as_ref(), layout.bounds(), &mut should_reset);
331
332        if should_reset.0 {
333            tree.state.downcast_mut::<State<P>>().should_reset = true;
334        }
335
336        self.element.as_widget_mut().operate(
337            &mut tree.children[0],
338            layout,
339            viewport,
340            renderer,
341            operation,
342        );
343    }
344
345    fn overlay<'a>(
346        &'a mut self,
347        tree: &'a mut Tree,
348        layout: Layout,
349        renderer: &Renderer,
350        viewport: &Rectangle,
351        translation: Vector,
352        window: Size,
353    ) -> Vec<overlay::Element<'a, Message, Theme, Renderer>> {
354        self.element.as_widget_mut().overlay(
355            &mut tree.children[0],
356            layout,
357            renderer,
358            viewport,
359            translation,
360            window,
361        )
362    }
363}

378struct State<P: Program> {
379    animation: P,
380    instant: Instant,
381    key: Key,
382    should_reset: bool,
383    size: Option<Size<Length>>,
384}

386#[derive(Default, Clone, Copy, Hash, PartialEq, Eq)]
387struct Key(u64);
388
389impl Key {
390    fn new(data: impl Hash) -> Self {
391        let mut hasher = DefaultHasher::new();
392        data.hash(&mut hasher);
393        Self(hasher.finish())
394    }
395}

399/// Reset the [`Animation`] of a [`Transition`].
400pub fn reset<Message>(id: impl Into<widget::Id>) -> iced_runtime::Task<Message>
401where
402    Message: iced_runtime::futures::MaybeSend + 'static,
403{
404    let id = id.into();
405    iced_runtime::task::widget(reset_raw(id)).discard()
406}
407
408/// An [`Operation`] to reset the [`Animation`] of a [`Transition`].
409pub fn reset_raw(id: impl Into<widget::Id>) -> impl Operation {
410    struct Reset(widget::Id);
411
412    impl Operation for Reset {
413        fn traverse(&mut self, operate: &mut dyn FnMut(&mut dyn Operation<()>)) {
414            operate(self);
415        }
416
417        fn custom(
418            &mut self,
419            id: Option<&widget::Id>,
420            _bounds: Rectangle,
421            state: &mut dyn std::any::Any,
422        ) {
423            if id == Some(&self.0)
424                && let Some(ShouldReset(should_reset)) = state.downcast_mut()
425            {
426                *should_reset = true;
427            }
428        }
429    }
430
431    Reset(id.into())
432}
