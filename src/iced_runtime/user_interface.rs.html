17/// A set of interactive graphical elements with a specific [`Layout`].
18///
19/// It can be updated and drawn.
20///
21/// Iced tries to avoid dictating how to write your event loop. You are in
22/// charge of using this type in your system in any way you want.
23///
24/// # Example
25/// The [`integration`] example uses a [`UserInterface`] to integrate Iced in an
26/// existing graphical application.
27///
28/// [`integration`]: https://github.com/iced-rs/iced/tree/master/examples/integration
29pub struct UserInterface<'a, Message, Theme, Renderer> {
30    root: widget::Element<'a, Message, Theme, Renderer>,
31    state: widget::Tree,
32    overlay: Option<mouse::Interaction>,
33    bounds: Size,
34    last_click: Option<mouse::Click>,
35    selection: Option<(Point, Point)>,
36}

44    /// Builds a user interface for a [`Widget`].
45    ///
46    /// It is able to avoid expensive computations when using a [`Cache`]
47    /// obtained from a previous instance of a [`UserInterface`].
48    ///
49    /// # Example
50    /// Imagine we want to build a [`UserInterface`] for
51    /// [the counter example that we previously wrote](index.html#usage). Here
52    /// is naive way to set up our application loop:
53    ///
54    /// ```no_run
55    /// # mod iced_wgpu {
56    /// #     pub type Renderer = ();
57    /// # }
58    /// #
59    /// # pub struct Counter;
60    /// #
61    /// # impl Counter {
62    /// #     pub fn new() -> Self { Counter }
63    /// #     pub fn view(&self) -> iced_core::Element<(), (), Renderer> { unimplemented!() }
64    /// #     pub fn update(&mut self, _: ()) {}
65    /// # }
66    /// use iced_runtime::core::shell;
67    /// use iced_runtime::core::window;
68    /// use iced_runtime::core::Size;
69    /// use iced_runtime::user_interface::{self, UserInterface};
70    /// use iced_wgpu::Renderer;
71    ///
72    /// // Initialization
73    /// let mut counter = Counter::new();
74    /// let mut cache = user_interface::Cache::new();
75    /// let mut renderer = Renderer::default();
76    /// let mut window = window::Headless; // This should be a proper window, like a `winit` one
77    /// let mut waker = shell::Waker::noop();
78    /// let mut window_size = Size::new(1024.0, 768.0);
79    ///
80    /// // Application loop
81    /// loop {
82    ///     // Process system events here...
83    ///
84    ///     // Build the user interface
85    ///     let user_interface = UserInterface::build(
86    ///         counter.view(),
87    ///         window_size,
88    ///         cache,
89    ///         &mut renderer,
90    ///     );
91    ///
92    ///     // Update and draw the user interface here...
93    ///     // ...
94    ///
95    ///     // Obtain the cache for the next iteration
96    ///     cache = user_interface.into_cache();
97    /// }
98    /// ```
99    pub fn build(
100        root: impl Widget<Message, Theme, Renderer> + 'a,
101        bounds: Size,
102        cache: Cache,
103        renderer: &mut Renderer,
104    ) -> Self {
105        let Cache {
106            mut state,
107            last_click,
108            selection,
109        } = cache;
110
111        let mut root = root._boxed();
112        state.diff(&mut root);
113
114        root.layout(
115            &mut state,
116            renderer,
117            &layout::Limits::new(Size::ZERO, bounds),
118        );
119
120        UserInterface {
121            root,
122            state,
123            overlay: None,
124            bounds,
125            last_click,
126            selection,
127        }
128    }

130    /// Updates the [`UserInterface`] by processing each provided [`Event`].
131    ///
132    /// It returns __messages__ that may have been produced as a result of user
133    /// interactions. You should feed these to your __update logic__.
134    ///
135    /// # Example
136    /// Let's allow our [counter](index.html#usage) to change state by
137    /// completing [the previous example](#example):
138    ///
139    /// ```no_run
140    /// # mod iced_wgpu {
141    /// #     pub type Renderer = ();
142    /// # }
143    /// #
144    /// # pub struct Counter;
145    /// #
146    /// # impl Counter {
147    /// #     pub fn new() -> Self { Counter }
148    /// #     pub fn view(&self) -> iced_core::Element<(), (), Renderer> { unimplemented!() }
149    /// #     pub fn update(&mut self, _: ()) {}
150    /// # }
151    /// use iced_runtime::core::mouse;
152    /// use iced_runtime::core::shell;
153    /// use iced_runtime::core::window;
154    /// use iced_runtime::core::Size;
155    /// use iced_runtime::user_interface::{self, UserInterface};
156    /// use iced_wgpu::Renderer;
157    ///
158    /// let mut counter = Counter::new();
159    /// let mut cache = user_interface::Cache::new();
160    /// let mut renderer = Renderer::default();
161    /// let mut window = window::Headless; // This should be a proper window, like a `winit` one
162    /// let mut waker = shell::Waker::noop();
163    /// let mut window_size = Size::new(1024.0, 768.0);
164    /// let mut cursor = mouse::Cursor::default();
165    ///
166    /// // Initialize our event storage
167    /// let mut events = Vec::new();
168    /// let mut messages = shell::Bus::new();
169    ///
170    /// loop {
171    ///     // Obtain system events...
172    ///
173    ///     let mut user_interface = UserInterface::build(
174    ///         counter.view(),
175    ///         window_size,
176    ///         cache,
177    ///         &mut renderer,
178    ///     );
179    ///
180    ///     // Update the user interface
181    ///     let (state, event_statuses) = user_interface.update(
182    ///         &window,
183    ///         &waker,
184    ///         &events,
185    ///         cursor,
186    ///         &mut renderer,
187    ///         &mut messages
188    ///     );
189    ///
190    ///     cache = user_interface.into_cache();
191    ///
192    ///     // Process the produced messages
193    ///     for (message, _receipt) in messages.drain() {
194    ///         counter.update(message);
195    ///     }
196    /// }
197    /// ```
198    pub fn update(
199        &mut self,
200        window: &dyn Window,
201        waker: &shell::Waker,
202        events: &[Event],
203        cursor: mouse::Cursor,
204        renderer: &mut Renderer,
205        messages: &mut shell::Bus<Message>,
206    ) -> (State, Vec<event::Status>) {
207        let mut status = shell::Invalidation::None;
208        let mut redraw_request = window::RedrawRequest::Wait;
209        let mut input_method = InputMethod::Disabled;
210        let mut clipboard = Clipboard::new();
211        let mut layout = Layout::new(self.state.size);
212        let viewport = Rectangle::with_size(self.bounds);
213
214        let overlay = self.root.overlay(
215            &mut self.state,
216            layout,
217            renderer,
218            &viewport,
219            Vector::ZERO,
220            self.bounds,
221        );
222
223        let mut maybe_overlay = (!overlay.is_empty()).then(|| overlay::Group::new(overlay));
224
225        let (base_cursor, overlay_statuses, overlay_interaction) = if maybe_overlay.is_some() {
226            let mut overlay = maybe_overlay.as_mut().unwrap();
227            let mut event_statuses = Vec::new();
228
229            for event in events {
230                let mut shell = Shell::new(window, waker.clone(), messages);
231
232                overlay.update(event, cursor, renderer, &mut shell);
233
234                let invalidation = shell.invalidation();
235                status = status.max(invalidation);
236                redraw_request = redraw_request.min(shell.redraw_request());
237
238                event_statuses.push(shell.event_status());
239                input_method.merge(shell.input_method());
240                clipboard.merge(shell.clipboard_mut());
241
242                if let shell::Invalidation::Layout(_) | shell::Invalidation::Overlay = invalidation
243                {
244                    drop(maybe_overlay);
245
246                    if let shell::Invalidation::Layout(diff) = invalidation {
247                        match diff {
248                            shell::Diff::Perform => {
249                                self.root.diff(&mut self.state);
250                            }
251                            shell::Diff::Skip => {}
252                        }
253
254                        self.root.layout(
255                            &mut self.state,
256                            renderer,
257                            &layout::Limits::new(Size::ZERO, self.bounds),
258                        );
259
260                        layout = Layout::new(self.state.size);
261                    }
262
263                    maybe_overlay = {
264                        let overlay = self.root.overlay(
265                            &mut self.state,
266                            layout,
267                            renderer,
268                            &viewport,
269                            Vector::ZERO,
270                            self.bounds,
271                        );
272
273                        (!overlay.is_empty()).then(|| overlay::Group::new(overlay))
274                    };
275
276                    if maybe_overlay.is_none() {
277                        event_statuses.resize(events.len(), event::Status::Ignored);
278                        break;
279                    }
280
281                    overlay = maybe_overlay.as_mut().unwrap();
282                }
283            }
284
285            let (base_cursor, interaction) = if let Some(overlay) = maybe_overlay.as_mut() {
286                let interaction = cursor
287                    .position()
288                    .map(|cursor_position| {
289                        overlay
290                            .mouse_interaction(mouse::Cursor::Available(cursor_position), renderer)
291                    })
292                    .unwrap_or_default();
293
294                if interaction == mouse::Interaction::None {
295                    (cursor, mouse::Interaction::None)
296                } else {
297                    (cursor.levitate(), interaction)
298                }
299            } else {
300                (cursor, mouse::Interaction::None)
301            };
302
303            self.overlay = maybe_overlay.as_ref().map(|_| interaction);
304
305            (base_cursor, event_statuses, interaction)
306        } else {
307            self.overlay = None;
308
309            (
310                cursor,
311                vec![event::Status::Ignored; events.len()],
312                mouse::Interaction::None,
313            )
314        };
315
316        drop(maybe_overlay);
317
318        let event_statuses = events
319            .iter()
320            .zip(overlay_statuses)
321            .map(|(event, overlay_status)| {
322                if matches!(overlay_status, event::Status::Captured) {
323                    return overlay_status;
324                }
325
326                let mut shell = Shell::new(window, waker.clone(), messages);
327
328                self.root.update(
329                    &mut self.state,
330                    event,
331                    layout,
332                    base_cursor,
333                    renderer,
334                    &mut shell,
335                    &viewport,
336                );
337
338                if shell.event_status() == event::Status::Captured {
339                    self.overlay = None;
340                }
341
342                let invalidation = shell.invalidation();
343
344                if let shell::Invalidation::Layout(_) | shell::Invalidation::Overlay = invalidation
345                {
346                    if let shell::Invalidation::Layout(diff) = invalidation {
347                        match diff {
348                            shell::Diff::Perform => {
349                                self.root.diff(&mut self.state);
350                            }
351                            shell::Diff::Skip => {}
352                        }
353
354                        self.root.layout(
355                            &mut self.state,
356                            renderer,
357                            &layout::Limits::new(Size::ZERO, self.bounds),
358                        );
359
360                        layout = Layout::new(self.state.size);
361                    }
362
363                    let overlay = self.root.overlay(
364                        &mut self.state,
365                        layout,
366                        renderer,
367                        &viewport,
368                        Vector::ZERO,
369                        self.bounds,
370                    );
371
372                    if !overlay.is_empty() {
373                        let mut overlay = overlay::Group::new(overlay);
374
375                        self.overlay = Some(overlay.mouse_interaction(cursor, renderer));
376                    }
377                }
378
379                match (&event, shell.event_status()) {
380                    (
381                        Event::Mouse(mouse::Event::ButtonPressed(mouse::Button::Left)),
382                        event::Status::Ignored,
383                    ) => {
384                        if let Some(position) = cursor.position() {
385                            self.last_click = Some(mouse::Click::new(
386                                position,
387                                mouse::Button::Left,
388                                self.last_click,
389                            ));
390
391                            self.selection = Some((position, Point::ORIGIN));
392                        }
393                    }
394                    (Event::Mouse(mouse::Event::ButtonReleased(mouse::Button::Left)), _) => {
395                        self.selection = None;
396                    }
397                    (
398                        Event::Keyboard(keyboard::Event::KeyPressed {
399                            key,
400                            modifiers,
401                            repeat: false,
402                            ..
403                        }),
404                        event::Status::Ignored,
405                    ) if matches!(key.as_ref(), keyboard::Key::Character("c"))
406                        && modifiers.control() =>
407                    {
408                        let mut copy = operation::text::copy();
409                        self.operate(renderer, &mut operation::black_box(&mut copy));
410
411                        if let operation::Outcome::Some(text) = copy.finish() {
412                            shell.write_clipboard(text);
413                        }
414                    }
415                    _ => {}
416                }
417
418                status = status.max(invalidation);
419                redraw_request = redraw_request.min(shell.redraw_request());
420                input_method.merge(shell.input_method());
421                clipboard.merge(shell.clipboard_mut());
422
423                shell.event_status()
424            })
425            .collect();
426
427        if let Some((start, end)) = self.selection.as_mut()
428            && let Some(click) = self.last_click
429            && let Some(position) = cursor.position()
430            && position != *end
431        {
432            *end = position;
433
434            let mut select = operation::text::select(
435                *start,
436                *end,
437                match click.kind() {
438                    mouse::click::Kind::Single => text::Target::Character,
439                    mouse::click::Kind::Double => text::Target::Word,
440                    mouse::click::Kind::Triple => text::Target::Line,
441                },
442            );
443
444            self.operate(renderer, &mut select);
445            redraw_request = window::RedrawRequest::NextFrame;
446        }
447
448        let mouse_interaction = if overlay_interaction == mouse::Interaction::None {
449            self.root
450                .mouse_interaction(&self.state, layout, base_cursor, &viewport, renderer)
451        } else {
452            overlay_interaction
453        };
454
455        let change = match status {
456            shell::Invalidation::None => Change::None,
457            shell::Invalidation::Overlay => Change::Overlay,
458            shell::Invalidation::Layout(_) => Change::Layout,
459            shell::Invalidation::Widgets => return (State::Outdated, event_statuses),
460        };
461
462        (
463            State::Updated {
464                mouse_interaction,
465                redraw_request,
466                input_method,
467                clipboard,
468                change,
469            },
470            event_statuses,
471        )
472    }

474    /// Draws the [`UserInterface`] with the provided [`Renderer`].
475    ///
476    /// It returns the current [`mouse::Interaction`]. You should update the
477    /// icon of the mouse cursor accordingly in your system.
478    ///
479    /// [`Renderer`]: crate::core::Renderer
480    ///
481    /// # Example
482    /// We can finally draw our [counter](index.html#usage) by
483    /// [completing the last example](#example-1):
484    ///
485    /// ```no_run
486    /// # mod iced_wgpu {
487    /// #     pub type Renderer = ();
488    /// #     pub type Theme = ();
489    /// # }
490    /// #
491    /// # pub struct Counter;
492    /// #
493    /// # impl Counter {
494    /// #     pub fn new() -> Self { Counter }
495    /// #     pub fn view(&self) -> Element<(), (), Renderer> { unimplemented!() }
496    /// #     pub fn update(&mut self, _: ()) {}
497    /// # }
498    /// use iced_runtime::core::mouse;
499    /// use iced_runtime::core::renderer;
500    /// use iced_runtime::core::shell;
501    /// use iced_runtime::core::window;
502    /// use iced_runtime::core::{Element, Size};
503    /// use iced_runtime::user_interface::{self, UserInterface};
504    /// use iced_wgpu::{Renderer, Theme};
505    ///
506    /// let mut counter = Counter::new();
507    /// let mut cache = user_interface::Cache::new();
508    /// let mut renderer = Renderer::default();
509    /// let mut window = window::Headless; // This should be a proper window, like a `winit` one
510    /// let mut waker = shell::Waker::noop();
511    /// let mut window_size = Size::new(1024.0, 768.0);
512    /// let mut cursor = mouse::Cursor::default();
513    /// let mut events = Vec::new();
514    /// let mut messages = shell::Bus::new();
515    /// let mut theme = Theme::default();
516    ///
517    /// loop {
518    ///     // Obtain system events...
519    ///
520    ///     let mut user_interface = UserInterface::build(
521    ///         counter.view(),
522    ///         window_size,
523    ///         cache,
524    ///         &mut renderer,
525    ///     );
526    ///
527    ///     // Update the user interface
528    ///     let event_statuses = user_interface.update(
529    ///         &window,
530    ///         &waker,
531    ///         &events,
532    ///         cursor,
533    ///         &mut renderer,
534    ///         &mut messages
535    ///     );
536    ///
537    ///     // Draw the user interface
538    ///     let mouse_interaction = user_interface.draw(&mut renderer, &theme, &renderer::Style::default(), cursor);
539    ///
540    ///     cache = user_interface.into_cache();
541    ///
542    ///     for (message, _receipt) in messages.drain() {
543    ///         counter.update(message);
544    ///     }
545    ///
546    ///     // Update mouse cursor icon...
547    ///     // Flush rendering operations...
548    /// }
549    /// ```
550    pub fn draw(
551        &mut self,
552        renderer: &mut Renderer,
553        theme: &Theme,
554        style: &renderer::Style,
555        cursor: mouse::Cursor,
556    ) {
557        let viewport = Rectangle::with_size(self.bounds);
558        renderer.reset(viewport);
559
560        let layout = Layout::new(self.state.size);
561        let base_cursor = match &self.overlay {
562            None | Some(mouse::Interaction::None) => cursor,
563            _ => cursor.levitate(),
564        };
565
566        self.root.draw(
567            &self.state,
568            renderer,
569            theme,
570            style,
571            layout,
572            base_cursor,
573            &viewport,
574        );
575
576        let Self { overlay, root, .. } = self;
577
578        if overlay.is_none() {
579            return;
580        }
581
582        let overlay = root.overlay(
583            &mut self.state,
584            layout,
585            renderer,
586            &viewport,
587            Vector::ZERO,
588            self.bounds,
589        );
590
591        if !overlay.is_empty() {
592            let mut overlay = overlay::Group::new(overlay);
593
594            overlay.draw(renderer, theme, style, cursor);
595        }
596    }

598    /// Applies a [`widget::Operation`] to the [`UserInterface`].
599    pub fn operate(&mut self, renderer: &Renderer, operation: &mut dyn widget::Operation) {
600        let viewport = Rectangle::with_size(self.bounds);
601        let layout = Layout::new(self.state.size);
602
603        operation.traverse(&mut |operation| {
604            self.root
605                .operate(&mut self.state, layout, &viewport, renderer, operation);
606        });
607
608        let overlay = self.root.overlay(
609            &mut self.state,
610            layout,
611            renderer,
612            &viewport,
613            Vector::ZERO,
614            self.bounds,
615        );
616
617        if !overlay.is_empty() {
618            let mut overlay = overlay::Group::new(overlay);
619
620            overlay.operate(renderer, operation);
621        }
622    }

624    /// Relayouts and returns a new  [`UserInterface`] using the provided
625    /// bounds.
626    pub fn relayout(self, bounds: Size, renderer: &mut Renderer) -> Self {
627        Self::build(
628            self.root,
629            bounds,
630            Cache {
631                state: self.state,
632                last_click: self.last_click,
633                selection: self.selection,
634            },
635            renderer,
636        )
637    }

639    /// Extract the [`Cache`] of the [`UserInterface`], consuming it in the
640    /// process.
641    pub fn into_cache(self) -> Cache {
642        Cache {
643            state: self.state,
644            last_click: self.last_click,
645            selection: self.selection,
646        }

678/// The resulting state after updating a [`UserInterface`].
679#[derive(Debug)]
680pub enum State {
681    /// The [`UserInterface`] is outdated and needs to be rebuilt.
682    Outdated,
683
684    /// The [`UserInterface`] is up-to-date and can be reused without
685    /// rebuilding.
686    Updated {
687        /// The current [`mouse::Interaction`] of the user interface.
688        mouse_interaction: mouse::Interaction,
689        /// The [`window::RedrawRequest`] describing when a redraw should be performed.
690        redraw_request: window::RedrawRequest,
691        /// The current [`InputMethod`] strategy of the user interface.
692        input_method: InputMethod,
693        /// The set of [`Clipboard`] requests that the user interface has produced.
694        clipboard: Clipboard,
695        /// The [`Change`] that the [`UserInterface`] experienced.
696        change: Change,
697    },
698}
699
700/// The change experienced by a [`UserInterface`] while being
701/// updated.
702#[derive(Debug, Clone, Copy, PartialEq, Eq)]
703pub enum Change {
704    /// Nothing was invalidated: the [`UserInterface`] is unchanged.
705    None,
706    /// Only the application overlays were recreated.
707    Overlay,
708    /// The application layout was recomputed.
709    Layout,
710}
