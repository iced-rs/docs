    Theme: Catalog,
    Renderer: Renderer,

Expand description

A bunch of text.

§Example

use iced::widget::text;
use iced::color;

enum Message {
    // ...
}

fn view(state: &State) -> Element<'_, Message> {
    text("Hello, this is iced!")
        .size(20)
        .color(color!(0x0000ff))
        .into()
}
